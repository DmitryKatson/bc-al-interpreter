// DEMO 9: Queries and Dataset Exploration
// Define a query object at runtime and render the result set in the console

query CustomerBalances
{
  elements
  {
    dataitem(Customer; Customer)
    {
      DataItemTableFilter = "Blocked" = filter(<> All);
      column(NoColumn; "No.")
      {
      }
      column(NameColumn; Name)
      {
      }
      column(BalanceLCY; "Balance (LCY)")
      {
        ColumnFilter = BalanceLCY = filter(<> 0);
        Method = Sum;
      }
    }
  }
}

begin
  Message('=== QUERY DEMO ===');
  ExecuteQuery('CustomerBalances');
end;

// Nested dataitem example

query SalesOrdersWithLines
{
  elements
  {
    dataitem(SalesHeader; "Sales Header")
    {
      DataItemTableFilter = "Document Type" = const(Order), "Sell-to Customer No." = const(10000);
      column(OrderNo; "No.")
      {
      }
      column(CustomerNo; "Sell-to Customer No.")
      {
      }

      dataitem(SalesLines; "Sales Line")
      {
        DataItemLink = "Document No." = SalesHeader."No.";
        DataItemTableFilter = "Document Type" = const(Order);
        column(LineNo; "Line No.")
        {
        }
        column(ItemNo; "No.")
        {
        }
        column(Quantity; Quantity)
        {
        }
      }
    }
  }
}

begin
  Message('=== SALES ORDERS WITH LINES ===');
  ExecuteQuery('SalesOrdersWithLines');
end;

// Bank account ledger linked to bank account

query BankLedgerWithAccount
{
  elements
  {
    dataitem(BankLedger; "Bank Account Ledger Entry")
    {
      DataItemTableFilter = Open = filter(= true);
      column(EntryNo; "Entry No.")
      {
      }
      column(AccountNo; "Bank Account No.")
      {
      }
      column(PostingDate; "Posting Date")
      {
      }
      column(AmountLCY; "Amount (LCY)")
      {
      }

      dataitem(BankAccount; "Bank Account")
      {
        DataItemLink = "No." = BankLedger."Bank Account No.";
        column(AccountName; Name)
        {
        }
        column(Currency; "Currency Code")
        {
        }
      }
    }
  }
}

begin
  Message('=== BANK LEDGER WITH ACCOUNT DETAILS ===');
  ExecuteQuery('BankLedgerWithAccount');
end;
